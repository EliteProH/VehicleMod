package com.elite.vehiclesmod;

import net.neoforged.fml.common.Mod;

import net.neoforged.fml.ModLoadingContext;

import net.minecraft.world.entity.MobCategory;

import net.neoforged.bus.api.IEventBus;

@Mod(VehiclesMod.MODID)
public class VehiclesMod {
    public static final String MODID = "vehiclesmod";

    // Register entities
    private static final DeferredRegister<EntityType<?>> ENTITIES =
        DeferredRegister.create(Registries.ENTITY_TYPE, MODID);

    public static final EntityType<Boat> PORSCHE_911_GT3_RS = EntityType.Builder.<Boat>of(Boat::new, MobCategory.MISC)
            .sized(1.5F, 0.6F)
            .build("porsche_911_gt3_rs");

    public VehiclesMod() {
        IEventBus modEventBus = ModLoadingContext.get().getActiveContainer().getEventBus(); // Alternative event bus access
        ENTITIES.register(modEventBus);

        modEventBus.addListener(this::setup);
        modEventBus.addListener(this::doClientStuff);
    }

    private void setup(final FMLCommonSetupEvent event) {
        // Register additional entities here
    }

    private void doClientStuff(final FMLClientSetupEvent event) {
        // Client-side rendering and input handling
    }
}
